cmake_minimum_required(VERSION 3.20)
project(IgniteComProtocol LANGUAGES CXX)
add_library(IgnitePacketProtocol STATIC PktMgr/PktMgr.cpp)
target_include_directories(IgnitePacketProtocol PUBLIC PktMgr)
target_compile_features(IgnitePacketProtocol PUBLIC cxx_std_17)
include(CTest)
if(BUILD_TESTING)
    add_executable(IgniteProtocolTests tests/protocol_tests.cpp)
    target_link_libraries(IgniteProtocolTests PRIVATE IgnitePacketProtocol)
    add_test(NAME IgniteProtocolTests COMMAND IgniteProtocolTests)
endif()
